ЧТО СДЕЛАНО
----------------------------------------------------------------
1. В решение WebApplication1.sln добавлен новый проект Blazor Web App:
   WebApplication1.Blazor.SSR.
   Команда создания:
     dotnet new blazor -n WebApplication1.Blazor.SSR --interactivity Server --all-interactive --framework net8.0
   Команда добавления в решение:
     dotnet sln WebApplication1.sln add WebApplication1.Blazor.SSR/WebApplication1.Blazor.SSR.csproj

2. Задание №1: изучены Program.cs, Components/App.razor, Components/Routes.razor,
   Components/Pages, Components/Layout/MainLayout.razor (макет и @Body),
   Components/Layout/NavMenu.razor (NavLink), Components/_Imports.razor.

3. Задание №2: проект использует интерактивный режим InteractiveServer.
   В Program.cs зарегистрированы AddRazorComponents().AddInteractiveServerComponents(),
   а конечные точки подключаются через MapRazorComponents<App>().AddInteractiveServerRenderMode().
   В App.razor компонент Routes имеет @rendermode="InteractiveServer", поэтому
   переходы между страницами выполняются через уже открытое WebSocket-соединение
   Blazor (blazor.web.js), без новых HTTP-запросов на сервер за HTML-страницами.

4. Задание №3: на странице Counter добавлены поле ввода и кнопка "Установить".
   Используется EditForm, InputNumber, DataAnnotationsValidator, ValidationSummary.
   Правило валидации описано в отдельном классе CounterFormModel через [Range(1, 10)].
   При OnValidSubmit значение из формы присваивается счетчику.

5. Задание №4: начальное значение счетчика можно передать в строке запроса.
   Поддерживаются оба варианта:
     /counter/200       - значение в сегменте маршрута (как на скриншоте задания);
     /counter?count=200 - значение в query string.
   Для маршрута объявлен @page "/counter/{InitialCount:int}", для query string
   используется параметр [SupplyParameterFromQuery(Name = "count")].

6. Во всех исходных файлах проекта удалены комментарии (C#, Razor, cshtml, JS, CSS).
   Сторонние файлы wwwroot/lib и wwwroot/bootstrap не изменялись.

КАК ПРОВЕРИТЬ
----------------------------------------------------------------
Запуск:
  dotnet run --project WebApplication1.Blazor.SSR/WebApplication1.Blazor.SSR.csproj

Открыть в браузере:
  /counter           -> Current count: 0
  /counter/200       -> Current count: 200
  /counter?count=7   -> Current count: 7

На странице Counter:
  - нажать "Click me" - счетчик увеличится на 1;
  - ввести число от 1 до 10 и нажать "Установить" - счетчик примет введенное значение;
  - ввести число вне диапазона (например, 12) или не число - появится сообщение
    "Значение должно быть целым числом от 1 до 10." / сообщение о неверном формате.

F12 -> Network: при переключении Home/Counter/Weather новые HTTP-запросы за страницами
не отправляются; F12 -> Console -> Messages: открыто WebSocket-соединение.

Program.cs
========================================================================
using WebApplication1.Blazor.SSR.Components;

var builder = WebApplication.CreateBuilder(args);


builder.Services.AddRazorComponents()
    .AddInteractiveServerComponents();

var app = builder.Build();


if (!app.Environment.IsDevelopment())
{
    app.UseExceptionHandler("/Error", createScopeForErrors: true);
    
    app.UseHsts();
}

app.UseHttpsRedirection();

app.UseStaticFiles();
app.UseAntiforgery();

app.MapRazorComponents<App>()
    .AddInteractiveServerRenderMode();

app.Run();

@page "/counter"
@page "/counter/{InitialCount:int}"
@using WebApplication1.Blazor.SSR.Models

<PageTitle>Counter</PageTitle>

<h1>Counter</h1>

<p role="status">Current count: @currentCount</p>

<button class="btn btn-primary" @onclick="IncrementCount">Click me</button>

<EditForm Model="model" OnValidSubmit="SetCount" class="mt-3">
    <DataAnnotationsValidator />
    <div class="mb-2">
        <InputNumber @bind-Value="model.Value" class="form-control" />
    </div>
    <button type="submit" class="btn btn-primary">Установить</button>
    <ValidationSummary />
</EditForm>

@code {
    private int currentCount;
    private readonly CounterFormModel model = new();

    [Parameter]
    public int InitialCount { get; set; }

    [SupplyParameterFromQuery(Name = "count")]
    public int? Count { get; set; }

    protected override void OnParametersSet()
    {
        if (Count.HasValue)
        {
            currentCount = Count.Value;
        }
        else
        {
            currentCount = InitialCount;
        }
    }

    private void IncrementCount()
    {
        currentCount++;
    }

    private void SetCount()
    {
        currentCount = model.Value;
    }
}

@page "/weather"

<PageTitle>Weather</PageTitle>

<h1>Weather</h1>

<p>This component demonstrates showing data.</p>

@if (forecasts == null)
{
    <p><em>Loading...</em></p>
}
else
{
    <table class="table">
        <thead>
            <tr>
                <th>Date</th>
                <th>Temp. (C)</th>
                <th>Temp. (F)</th>
                <th>Summary</th>
            </tr>
        </thead>
        <tbody>
            @foreach (var forecast in forecasts)
            {
                <tr>
                    <td>@forecast.Date.ToShortDateString()</td>
                    <td>@forecast.TemperatureC</td>
                    <td>@forecast.TemperatureF</td>
                    <td>@forecast.Summary</td>
                </tr>
            }
        </tbody>
    </table>
}

@code {
    private WeatherForecast[]? forecasts;

    protected override async Task OnInitializedAsync()
    {
        await Task.Delay(500);

        var startDate = DateOnly.FromDateTime(DateTime.Now);
        var summaries = new[] { "Freezing", "Bracing", "Chilly", "Cool", "Mild", "Warm", "Balmy", "Hot", "Sweltering", "Scorching" };
        forecasts = Enumerable.Range(1, 5).Select(index => new WeatherForecast
        {
            Date = startDate.AddDays(index),
            TemperatureC = Random.Shared.Next(-20, 55),
            Summary = summaries[Random.Shared.Next(summaries.Length)]
        }).ToArray();
    }

    private class WeatherForecast
    {
        public DateOnly Date { get; set; }
        public int TemperatureC { get; set; }
        public string? Summary { get; set; }
        public int TemperatureF => 32 + (int)(TemperatureC / 0.5556);
    }
}
