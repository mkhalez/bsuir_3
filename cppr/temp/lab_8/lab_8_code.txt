==================== 1. Controllers/ProductController.cs — маршруты ====================
----- БЫЛО -----

{

----- СТАЛО (добавлены 2 строки атрибутов) -----

{
    ...

    [Route("Catalog/{category?}")]

Итоговые адреса: /Catalog и /Catalog/{category}, pageNo — через query (?pageNo=N).
Старый адрес /Product/Index больше не работает (действие на атрибут-маршрутах
исключено из conventional-маршрутизации). Все ссылки asp-controller=Product
(меню, пейджер, dropdown категорий) теперь генерируют /Catalog/... автоматически.

==================== 7. labs.API/EndPoints/DishEndpoints.cs — кэширование списка ====================
----- БЫЛО -----
        group.MapGet("/{category:alpha?}", async (
            IMediator mediator,
            string? category,
            int pageNo = 1,
            int pageSize = 3) =>
        {
            var data = await mediator.Send(
                new GetListOfProducts(category, pageNo, pageSize));
            return TypedResults.Ok(data);
        })
        .WithName("GetAllDishes")
        .AllowAnonymous();;

----- СТАЛО -----
        group.MapGet("/{category:alpha?}", async (
            IMediator mediator,
            HybridCache cache,
            string? category,
            int pageNo = 1,
            int pageSize = 3) =>
        {
            var data = await cache.GetOrCreateAsync(
                $"dishes_{category}_{pageNo}_{pageSize}",
                async token => await mediator.Send(
                    new GetListOfProducts(category, pageNo, pageSize), token),
                options: new HybridCacheEntryOptions
                {
                    Expiration = TimeSpan.FromMinutes(1),
                    LocalCacheExpiration = TimeSpan.FromSeconds(30)
                });
            return TypedResults.Ok(data);
        })
        .WithName("GetAllDishes")
        .AllowAnonymous();

Логика времен: запрос чаще чем раз в 30 сек — ответ из локального кэша,
от 30 сек до 1 мин — из Redis, реже чем раз в минуту — из базы.

==================== 9. labs.Domain/Models/Cart.cs — НОВЫЙ ====================
using labs.Domain.Entities;

namespace labs.Domain.Models;

public class Cart
{
    public Dictionary<int, CartItem> CartItems { get; set; } = new();

    public virtual void AddToCart(Dish dish)
    {
        if (CartItems.TryGetValue(dish.Id, out var item))
            item.Count++;
        else
            CartItems[dish.Id] = new CartItem { Item = dish, Count = 1 };
    }

    public virtual void RemoveItems(int id)
    {
        CartItems.Remove(id);
    }

    public virtual void ClearAll()
    {
        CartItems.Clear();
    }

    public int Count => CartItems.Values.Sum(item => item.Count);

    public double TotalCalories =>
        CartItems.Values.Sum(item => (double)item.Item.Calories * item.Count);
}

==================== 10. labs/Extensions/SessionExtensions.cs — НОВЫЙ ====================
using System.Text.Json;

namespace labs.Extensions;

public static class SessionExtensions
{
    public static void Set<T>(this ISession session, string key, T value)
    {
        session.SetString(key, JsonSerializer.Serialize(value));
    }

    public static T? Get<T>(this ISession session, string key)
    {
        var data = session.GetString(key);
        return data is null ? default : JsonSerializer.Deserialize<T>(data);
    }
}

==================== 12. labs/Models/SessionCart.cs — НОВЫЙ ====================
using System.Text.Json.Serialization;
using labs.Domain.Entities;
using labs.Domain.Models;
using labs.Extensions;

namespace labs.Models;

public class SessionCart : Cart
{
    [JsonIgnore]
    public ISession? Session { get; set; }

    public static Cart GetCart(IServiceProvider services)
    {
        var session = services
            .GetRequiredService<IHttpContextAccessor>()
            .HttpContext?.Session;
        var cart = session?.Get<SessionCart>("cart") ?? new SessionCart();
        cart.Session = session;
        return cart;
    }

    public override void AddToCart(Dish dish)
    {
        base.AddToCart(dish);
        Session?.Set("cart", this);
    }

    public override void RemoveItems(int id)
    {
        base.RemoveItems(id);
        Session?.Set("cart", this);
    }

    public override void ClearAll()
    {
        base.ClearAll();
        Session?.Remove("cart");
    }
}

==================== 14. labs/Controllers/CartController.cs — НОВЫЙ ====================
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;

namespace labs.Controllers;

public class CartController : Controller
{
    private readonly IProductService _productService;
    private readonly Cart _cart;

    public CartController(IProductService productService, Cart cart)
    {
        _productService = productService;
        _cart = cart;
    }

    public IActionResult Index()
    {
        return View(_cart);
    }

    [Authorize]
    [Route("[controller]/add/{id:int}")]
    public async Task<ActionResult> Add(int id, string? returnUrl)
    {
        var data = await _productService.GetProductByIdAsync(id);
        if (data.Successfull && data.Data is not null)
        {
            _cart.AddToCart(data.Data);
        }

        return Redirect(string.IsNullOrEmpty(returnUrl) ? "/Catalog" : returnUrl);
    }

    public IActionResult Remove(int id)
    {
        _cart.RemoveItems(id);
        return RedirectToAction("Index");
    }

    public IActionResult Clear()
    {
        _cart.ClearAll();
        return RedirectToAction("Index");
    }
}
